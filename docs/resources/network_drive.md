# twc_network_drive (Resource)

Resource for describing needed network drive and provides actual information about its status

## Примеры использования

```terraform
### Example network drive with preset

# Select a network drive preset of the required type (nvme/hdd)
data "twc_network_drive_preset" "example-preset" {
  type = "nvme"
}

resource "twc_network_drive" "example-drive" {
  name      = "Example network drive"
  comment   = "Some network drive"
  preset_id = data.twc_network_drive_preset.example-preset.id
}

### Example network drive with explicit size (without preset)

resource "twc_network_drive" "example-drive-sized" {
  name    = "Example sized drive"
  comment = "Some sized network drive"
  size    = 20 # disk size in GB
}

### Example network drive mounted to a server

data "twc_configurator" "example-configurator" {
  location = "ru-1"
}

data "twc_os" "example-os" {
  name    = "ubuntu"
  version = "22.04"
}

resource "twc_server" "example-server" {
  name  = "Example server"
  os_id = data.twc_os.example-os.id

  configuration {
    configurator_id = data.twc_configurator.example-configurator.id
    disk            = 1024 * 10
    cpu             = 1
    ram             = 1024
  }
}

resource "twc_network_drive" "example-drive-mounted" {
  name    = "Example mounted drive"
  comment = "Network drive mounted to server"
  size    = 20

  # Mount the drive to a resource (only `server` is currently supported)
  resource {
    type = "server"
    id   = twc_server.example-server.id
  }
}
```
