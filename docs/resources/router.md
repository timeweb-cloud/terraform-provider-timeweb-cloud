# twc_router (Resource)

Resource for describing virtual router

## Примеры использования

```terraform
# Select a router preset by price
data "twc_router_preset" "example-router-preset" {
  location = "ru-1"

  price_filter {
    from = 0
    to   = 1000
  }
}

# VPC network that the router will manage
resource "twc_vpc" "example-vpc" {
  name        = "Example VPC"
  description = "Some example VPC"
  subnet_v4   = "192.168.0.0/24"
  location    = "ru-1"
}

# Virtual router managing a single network with DHCP and SNAT enabled
resource "twc_router" "example-router" {
  name      = "example-router"
  comment   = "Some example router"
  preset_id = data.twc_router_preset.example-router-preset.id

  # networks block — TypeSet: list all networks managed by the router.
  # id holds the VPC identifier (pvc).
  networks {
    id = twc_vpc.example-vpc.id
    # DHCP for this network (set by a separate call after the network is attached)
    is_dhcp_enabled = true
    # Source NAT (SNAT) IPv4 for the network's outbound traffic (optional).
    # Remove the line to disable SNAT for the network.
    nat_ip = "203.0.113.1"
  }
}
```
