# twc_lb_rule (Resource)

Resource for describing needed load balancer rule and provides actual information about its status

## Примеры использования

```terraform
# Select a load balancer preset for ~10K RPS priced between 100 and 200 rub.
data "twc_lb_preset" "example-lb-preset" {
  requests_per_second = "10K"

  price_filter {
    from = 100
    to   = 200
  }
}

# Create the load balancer that we will add rules to
resource "twc_lb" "example-lb" {
  name = "example-lb"

  algo = "roundrobin"

  preset_id = data.twc_lb_preset.example-lb-preset.id
}

# Balancing rule: protocol/port at the balancer ingress -> backend protocol/port
resource "twc_lb_rule" "example-lb-rule" {
  lb_id = twc_lb.example-lb.id

  balancer_proto = "http2" # http | http2 | https | tcp
  balancer_port  = 83
  server_proto   = "http"
  server_port    = 82
}
```
