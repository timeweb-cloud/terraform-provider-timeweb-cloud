# twc_s3_bucket (Resource)

Resource for describing needed S3 bucket and provides actual information about its status

## Примеры использования

```terraform
### Example 1: hot bucket via preset

# In production S3 is available only in the ru-1 location
data "twc_s3_preset" "example-s3-hot-preset" {
  location = "ru-1"
  # storage_class = "hot" — regular (hot) bucket
  storage_class = "hot"

  disk = 10 * 1024
}

# Private hot bucket on the selected preset
resource "twc_s3_bucket" "example-s3-hot-bucket" {
  name      = "example-s3-hot-bucket"
  type      = "private" # access type: private, public, photos
  preset_id = data.twc_s3_preset.example-s3-hot-preset.id

  description = "Hot bucket for the example"

  # Automatically upgrade the preset when the current one overflows
  is_allow_auto_upgrade = true
}

### Example 2: cold bucket via preset

# Cold storage is selected with a preset where storage_class = "cold".
# In production S3 works only via presets (configurators are unavailable for S3).
data "twc_s3_preset" "example-s3-cold-preset" {
  location      = "ru-1"
  storage_class = "cold"

  disk = 10 * 1024
}

resource "twc_s3_bucket" "example-s3-cold-bucket" {
  name      = "example-s3-cold-bucket"
  type      = "private"
  preset_id = data.twc_s3_preset.example-s3-cold-preset.id
}

### Example 2b: selecting a plan via a configurator (configuration)
#
# The schema also supports a configuration {} block (mutually exclusive with preset_id),
# but in production configurators for S3 are usually unavailable — prefer preset_id.
#
# resource "twc_s3_bucket" "example-s3-configured-bucket" {
#   name = "example-s3-configured-bucket"
#   type = "private"
#
#   configuration {
#     configurator_id = data.twc_s3_configurator.example.id
#     disk            = 2048 # disk size in GB
#   }
# }

### Example 3: public bucket with a static website

resource "twc_s3_bucket" "example-s3-website-bucket" {
  name      = "example-s3-website-bucket"
  type      = "public"
  preset_id = data.twc_s3_preset.example-s3-hot-preset.id

  # Hosting a static website from the bucket
  website_config {
    enabled    = true
    index_page = "index.html"

    error_pages {
      code     = 404
      document = "404.html"
    }
  }

  # Bucket size limit in MB (only for pay-as-you-go presets).
  # Must be a multiple of 1024; 0 — no limit.
  max_size_mb = 10 * 1024
}
```
