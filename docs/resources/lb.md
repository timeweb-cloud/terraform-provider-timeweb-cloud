# twc_lb (Resource)

Resource for describing needed load balancer and provides actual information about its status

## Примеры использования

```terraform
# Select a load balancer preset for ~10K RPS priced between 100 and 200 rub.
data "twc_lb_preset" "example-lb-preset" {
  requests_per_second = "10K"

  price_filter {
    from = 100
    to   = 200
  }
}

resource "twc_project" "example-project" {
  name = "Example"
}

# Create the load balancer
resource "twc_lb" "example-lb" {
  name = "example-lb"

  algo = "roundrobin" # roundrobin | leastconn

  is_sticky    = false # pin the session to a single backend
  is_use_proxy = false # PROXY protocol to the backend
  is_ssl       = false # automatic HTTP -> HTTPS redirect
  is_keepalive = false # keep-alive to the backend

  health_check {
    proto = "http" # http | http2 | https | tcp

    port = 80

    path = "/health"

    inter   = 10 # check interval, sec
    timeout = 5  # check timeout, sec
    fall    = 3  # error threshold to mark backend down
    rise    = 2  # success threshold to mark backend up
  }

  # List of backend IPs
  ips = []

  preset_id = data.twc_lb_preset.example-lb-preset.id

  project_id = twc_project.example-project.id
}

# Balancing rule: incoming balancer traffic -> backend
resource "twc_lb_rule" "example-lb-rule" {
  lb_id = twc_lb.example-lb.id

  balancer_proto = "http2"
  balancer_port  = 83
  server_proto   = "http"
  server_port    = 82
}
```
